-- PCFORGE — TOÀN BỘ SQL MYSQL
-- Công nghệ: MySQL + Spring Boot + React.
-- Yêu cầu MySQL >= 8.0.16; khuyến nghị MySQL 8.4 LTS.
-- Chạy một lần trên database pcforge trống.
-- Phần 1–3: tạo database, schema và dữ liệu nền.
-- Phần 4–5: SQL tùy chọn nằm trong chú thích, không tự chạy.
-- Schema đã được kiểm tra tĩnh, chưa thực thi trực tiếp trên MySQL server.

-- ================================================================
-- PHẦN 1: bootstrap.sql
-- ================================================================
-- MySQL >= 8.0.16; recommended: MySQL 8.4 LTS.
-- Run once with an account allowed to create databases.
CREATE DATABASE IF NOT EXISTS pcforge
  CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;


USE pcforge;

-- ================================================================
-- PHẦN 2: V1__schema.sql
-- ================================================================
-- PCForge / MySQL 8.0.16+ / InnoDB / Spring Boot.
-- Select database pcforge before running. No DROP, no sample passwords.
-- All DATETIME values represent UTC. Money is DECIMAL, Java uses BigDecimal.
-- Execute once on an empty database, or let Flyway apply this migration.
SET NAMES utf8mb4;
SET time_zone = '+00:00';

-- ================================================================
-- 01. IDENTITY / RBAC
-- ================================================================
CREATE TABLE users (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  email VARCHAR(254) NOT NULL,
  phone VARCHAR(25),
  password_hash VARCHAR(255),
  full_name VARCHAR(150) NOT NULL,
  avatar_url VARCHAR(1000),
  status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
  email_verified_at DATETIME(6),
  last_login_at DATETIME(6),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  version BIGINT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_users_email (email),
  CONSTRAINT ck_users_status CHECK (status IN ('ACTIVE','BLOCKED','DEACTIVATED'))
) ENGINE=InnoDB;

CREATE TABLE auth_identities (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  provider VARCHAR(20) NOT NULL,
  provider_subject VARCHAR(255) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  UNIQUE KEY uk_identity_subject (provider, provider_subject),
  FOREIGN KEY (user_id) REFERENCES users(id),
  CHECK (provider IN ('GOOGLE','FACEBOOK'))
) ENGINE=InnoDB;

CREATE TABLE auth_tokens (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  token_hash BINARY(32) NOT NULL,
  purpose VARCHAR(30) NOT NULL,
  expires_at DATETIME(6) NOT NULL,
  used_at DATETIME(6),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  UNIQUE KEY uk_auth_token (token_hash),
  KEY ix_auth_token_expiry (expires_at),
  FOREIGN KEY (user_id) REFERENCES users(id),
  CHECK (purpose IN ('VERIFY_EMAIL','RESET_PASSWORD'))
) ENGINE=InnoDB;

CREATE TABLE auth_sessions (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  refresh_token_hash BINARY(32) NOT NULL,
  expires_at DATETIME(6) NOT NULL,
  revoked_at DATETIME(6),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  UNIQUE KEY uk_refresh_token (refresh_token_hash),
  KEY ix_session_expiry (expires_at),
  FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB;

CREATE TABLE roles (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(40) NOT NULL UNIQUE,
  name VARCHAR(100) NOT NULL
) ENGINE=InnoDB;
CREATE TABLE permissions (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(80) NOT NULL UNIQUE,
  description VARCHAR(255)
) ENGINE=InnoDB;
CREATE TABLE user_roles (
  user_id BIGINT NOT NULL,
  role_id BIGINT NOT NULL,
  PRIMARY KEY (user_id, role_id),
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (role_id) REFERENCES roles(id)
) ENGINE=InnoDB;
CREATE TABLE role_permissions (
  role_id BIGINT NOT NULL,
  permission_id BIGINT NOT NULL,
  PRIMARY KEY (role_id, permission_id),
  FOREIGN KEY (role_id) REFERENCES roles(id),
  FOREIGN KEY (permission_id) REFERENCES permissions(id)
) ENGINE=InnoDB;

CREATE TABLE customer_addresses (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  recipient_name VARCHAR(150) NOT NULL,
  phone VARCHAR(25) NOT NULL,
  country_code CHAR(2) NOT NULL DEFAULT 'VN',
  province_code VARCHAR(30),
  province_name VARCHAR(100) NOT NULL,
  district_code VARCHAR(30),
  district_name VARCHAR(100),
  ward_code VARCHAR(30),
  ward_name VARCHAR(100),
  address_line VARCHAR(500) NOT NULL,
  postal_code VARCHAR(20),
  is_default BOOLEAN NOT NULL DEFAULT FALSE,
  default_owner BIGINT GENERATED ALWAYS AS (CASE WHEN is_default = 1 THEN user_id ELSE NULL END) STORED,
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  UNIQUE KEY uk_address_default (default_owner),
  FOREIGN KEY (user_id) REFERENCES users(id),
  CHECK (is_default IN (0,1))
) ENGINE=InnoDB;
CREATE TABLE customer_groups (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(50) NOT NULL UNIQUE,
  name VARCHAR(150) NOT NULL
) ENGINE=InnoDB;
CREATE TABLE customer_group_members (
  group_id BIGINT NOT NULL,
  user_id BIGINT NOT NULL,
  PRIMARY KEY (group_id, user_id),
  FOREIGN KEY (group_id) REFERENCES customer_groups(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB;

-- ================================================================
-- 02. CATALOG / SKU / FACETED SEARCH
-- ================================================================
CREATE TABLE brands (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(150) NOT NULL UNIQUE,
  slug VARCHAR(180) NOT NULL UNIQUE,
  logo_url VARCHAR(1000),
  is_active BOOLEAN NOT NULL DEFAULT TRUE
) ENGINE=InnoDB;
CREATE TABLE categories (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  parent_id BIGINT,
  name VARCHAR(150) NOT NULL,
  slug VARCHAR(180) NOT NULL UNIQUE,
  category_group VARCHAR(20) NOT NULL,
  component_type VARCHAR(30),
  sort_order INT NOT NULL DEFAULT 0,
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  FOREIGN KEY (parent_id) REFERENCES categories(id),
  CHECK (category_group IN ('COMPONENT','ACCESSORY')),
  CHECK (component_type IS NULL OR component_type IN ('CPU','MOTHERBOARD','RAM','GPU','STORAGE','PSU','CASE','COOLER','FAN','MONITOR','OTHER'))
) ENGINE=InnoDB;
CREATE TABLE products (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  brand_id BIGINT NOT NULL,
  category_id BIGINT NOT NULL,
  name VARCHAR(255) NOT NULL,
  slug VARCHAR(280) NOT NULL UNIQUE,
  short_description VARCHAR(1000),
  description LONGTEXT,
  status VARCHAR(20) NOT NULL DEFAULT 'DRAFT',
  seo_title VARCHAR(255),
  seo_description VARCHAR(500),
  deleted_at DATETIME(6),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  KEY ix_product_listing (category_id, status, created_at),
  FULLTEXT KEY ft_product_search (name, short_description),
  FOREIGN KEY (brand_id) REFERENCES brands(id),
  FOREIGN KEY (category_id) REFERENCES categories(id),
  CHECK (status IN ('DRAFT','ACTIVE','ARCHIVED'))
) ENGINE=InnoDB;
CREATE TABLE product_variants (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  product_id BIGINT NOT NULL,
  sku VARCHAR(80) CHARACTER SET ascii COLLATE ascii_bin NOT NULL UNIQUE,
  barcode VARCHAR(100) CHARACTER SET ascii COLLATE ascii_bin UNIQUE,
  name VARCHAR(255) NOT NULL,
  list_price DECIMAL(19,2) NOT NULL,
  sale_price DECIMAL(19,2) NOT NULL,
  currency_code CHAR(3) NOT NULL DEFAULT 'VND',
  tax_rate DECIMAL(5,2) NOT NULL DEFAULT 0,
  weight_g DECIMAL(12,2) NOT NULL DEFAULT 0,
  package_length_mm DECIMAL(10,2),
  package_width_mm DECIMAL(10,2),
  package_height_mm DECIMAL(10,2),
  track_serial BOOLEAN NOT NULL DEFAULT FALSE,
  status VARCHAR(20) NOT NULL DEFAULT 'DRAFT',
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  version BIGINT NOT NULL DEFAULT 0,
  KEY ix_variant_price (status, sale_price),
  FOREIGN KEY (product_id) REFERENCES products(id),
  CHECK (list_price >= 0 AND sale_price >= 0 AND weight_g >= 0),
  CHECK (tax_rate BETWEEN 0 AND 100),
  CHECK (status IN ('DRAFT','ACTIVE','DISCONTINUED'))
) ENGINE=InnoDB;
CREATE TABLE product_media (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  product_id BIGINT NOT NULL,
  variant_id BIGINT,
  media_type VARCHAR(10) NOT NULL DEFAULT 'IMAGE',
  url VARCHAR(1000) NOT NULL,
  alt_text VARCHAR(255),
  sort_order INT NOT NULL DEFAULT 0,
  FOREIGN KEY (product_id) REFERENCES products(id),
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  CHECK (media_type IN ('IMAGE','VIDEO'))
) ENGINE=InnoDB;
CREATE TABLE attribute_definitions (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(80) NOT NULL UNIQUE,
  name VARCHAR(150) NOT NULL,
  data_type VARCHAR(10) NOT NULL,
  unit VARCHAR(30),
  CHECK (data_type IN ('TEXT','NUMBER','BOOLEAN','OPTION'))
) ENGINE=InnoDB;
CREATE TABLE attribute_options (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  attribute_id BIGINT NOT NULL,
  value VARCHAR(150) NOT NULL,
  sort_order INT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_attribute_option (attribute_id, value),
  UNIQUE KEY uk_attribute_option_pair (attribute_id, id),
  FOREIGN KEY (attribute_id) REFERENCES attribute_definitions(id)
) ENGINE=InnoDB;
CREATE TABLE category_attributes (
  category_id BIGINT NOT NULL,
  attribute_id BIGINT NOT NULL,
  is_required BOOLEAN NOT NULL DEFAULT FALSE,
  is_filterable BOOLEAN NOT NULL DEFAULT TRUE,
  sort_order INT NOT NULL DEFAULT 0,
  PRIMARY KEY (category_id, attribute_id),
  FOREIGN KEY (category_id) REFERENCES categories(id),
  FOREIGN KEY (attribute_id) REFERENCES attribute_definitions(id)
) ENGINE=InnoDB;
CREATE TABLE variant_attribute_values (
  variant_id BIGINT NOT NULL,
  attribute_id BIGINT NOT NULL,
  text_value VARCHAR(500),
  number_value DECIMAL(18,4),
  boolean_value BOOLEAN,
  option_id BIGINT,
  PRIMARY KEY (variant_id, attribute_id),
  KEY ix_facet_number (attribute_id, number_value),
  KEY ix_facet_option (attribute_id, option_id),
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (attribute_id) REFERENCES attribute_definitions(id),
  FOREIGN KEY (attribute_id, option_id) REFERENCES attribute_options(attribute_id, id),
  CHECK ((text_value IS NOT NULL) + (number_value IS NOT NULL) + (boolean_value IS NOT NULL) + (option_id IS NOT NULL) = 1),
  CHECK (boolean_value IS NULL OR boolean_value IN (0,1))
) ENGINE=InnoDB;
CREATE TABLE product_price_history (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  variant_id BIGINT NOT NULL,
  old_price DECIMAL(19,2) NOT NULL,
  new_price DECIMAL(19,2) NOT NULL,
  changed_by BIGINT,
  reason VARCHAR(500),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  KEY ix_price_history (variant_id, created_at),
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (changed_by) REFERENCES users(id),
  CHECK (old_price >= 0 AND new_price >= 0)
) ENGINE=InnoDB;
CREATE TABLE product_relations (
  product_id BIGINT NOT NULL,
  related_product_id BIGINT NOT NULL,
  relation_type VARCHAR(20) NOT NULL,
  PRIMARY KEY (product_id, related_product_id, relation_type),
  FOREIGN KEY (product_id) REFERENCES products(id),
  FOREIGN KEY (related_product_id) REFERENCES products(id),
  CHECK (product_id <> related_product_id),
  CHECK (relation_type IN ('RELATED','ACCESSORY','ALTERNATIVE'))
) ENGINE=InnoDB;

-- ================================================================
-- 03. TYPED COMPONENT SPECS / COMPATIBILITY
-- NULL = unknown, never automatically compatible.
-- ================================================================
CREATE TABLE spec_sources (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  source_url VARCHAR(1000),
  verified_at DATETIME(6),
  verified_by BIGINT,
  notes TEXT,
  FOREIGN KEY (verified_by) REFERENCES users(id)
) ENGINE=InnoDB;
CREATE TABLE cpu_sockets (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(40) NOT NULL UNIQUE,
  manufacturer VARCHAR(80) NOT NULL
) ENGINE=InnoDB;
CREATE TABLE cpu_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  socket_id BIGINT NOT NULL,
  family VARCHAR(100),
  generation VARCHAR(60),
  core_count SMALLINT,
  thread_count SMALLINT,
  base_power_w DECIMAL(8,2),
  max_power_w DECIMAL(8,2),
  has_igpu BOOLEAN,
  igpu_model VARCHAR(100),
  includes_cooler BOOLEAN,
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (socket_id) REFERENCES cpu_sockets(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (core_count > 0 AND thread_count > 0),
  CHECK (base_power_w >= 0 AND max_power_w >= 0)
) ENGINE=InnoDB;
CREATE TABLE cpu_memory_support (
  cpu_variant_id BIGINT NOT NULL,
  memory_type VARCHAR(10) NOT NULL,
  max_capacity_gb INT,
  max_official_speed_mts INT,
  PRIMARY KEY (cpu_variant_id, memory_type),
  FOREIGN KEY (cpu_variant_id) REFERENCES cpu_specs(variant_id),
  CHECK (max_capacity_gb > 0 AND max_official_speed_mts > 0)
) ENGINE=InnoDB;
CREATE TABLE motherboard_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  socket_id BIGINT NOT NULL,
  chipset VARCHAR(60) NOT NULL,
  form_factor VARCHAR(30) NOT NULL,
  memory_type VARCHAR(10) NOT NULL,
  memory_module_type VARCHAR(20) NOT NULL DEFAULT 'UDIMM',
  ram_slots SMALLINT NOT NULL,
  max_ram_gb INT NOT NULL,
  supports_ecc BOOLEAN,
  has_display_output BOOLEAN,
  has_wifi BOOLEAN,
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (socket_id) REFERENCES cpu_sockets(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (ram_slots > 0 AND max_ram_gb > 0)
) ENGINE=InnoDB;
CREATE TABLE motherboard_cpu_support (
  motherboard_variant_id BIGINT NOT NULL,
  cpu_variant_id BIGINT NOT NULL,
  minimum_bios VARCHAR(80),
  support_status VARCHAR(20) NOT NULL,
  source_id BIGINT,
  PRIMARY KEY (motherboard_variant_id, cpu_variant_id),
  FOREIGN KEY (motherboard_variant_id) REFERENCES motherboard_specs(variant_id),
  FOREIGN KEY (cpu_variant_id) REFERENCES cpu_specs(variant_id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (support_status IN ('SUPPORTED','UNSUPPORTED','UNKNOWN'))
) ENGINE=InnoDB;
CREATE TABLE ram_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  memory_type VARCHAR(10) NOT NULL,
  module_type VARCHAR(20) NOT NULL,
  capacity_per_module_gb INT NOT NULL,
  modules_per_kit SMALLINT NOT NULL DEFAULT 1,
  speed_mts INT NOT NULL,
  is_ecc BOOLEAN NOT NULL DEFAULT FALSE,
  is_registered BOOLEAN NOT NULL DEFAULT FALSE,
  height_mm DECIMAL(8,2),
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (capacity_per_module_gb > 0 AND modules_per_kit > 0 AND speed_mts > 0)
) ENGINE=InnoDB;
CREATE TABLE gpu_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  chipset VARCHAR(100) NOT NULL,
  vram_gb INT NOT NULL,
  length_mm DECIMAL(8,2),
  height_mm DECIMAL(8,2),
  thickness_slots DECIMAL(4,1),
  board_power_w DECIMAL(8,2),
  recommended_psu_w INT,
  pcie_generation SMALLINT,
  pcie_lanes SMALLINT,
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (vram_gb > 0 AND length_mm > 0 AND board_power_w >= 0)
) ENGINE=InnoDB;
CREATE TABLE storage_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  drive_type VARCHAR(10) NOT NULL,
  capacity_gb INT NOT NULL,
  interface_type VARCHAR(30) NOT NULL,
  protocol VARCHAR(30) NOT NULL,
  form_factor VARCHAR(30) NOT NULL,
  length_mm DECIMAL(8,2),
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (drive_type IN ('SSD','HDD')),
  CHECK (capacity_gb > 0)
) ENGINE=InnoDB;
CREATE TABLE psu_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  continuous_power_w INT NOT NULL,
  form_factor VARCHAR(30) NOT NULL,
  length_mm DECIMAL(8,2),
  efficiency_rating VARCHAR(40),
  atx_version VARCHAR(20),
  modular_type VARCHAR(20),
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (continuous_power_w > 0 AND length_mm > 0)
) ENGINE=InnoDB;
CREATE TABLE case_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  max_gpu_length_mm DECIMAL(8,2),
  max_cooler_height_mm DECIMAL(8,2),
  max_psu_length_mm DECIMAL(8,2),
  expansion_slots SMALLINT,
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (max_gpu_length_mm > 0 AND max_cooler_height_mm > 0 AND max_psu_length_mm > 0)
) ENGINE=InnoDB;
CREATE TABLE cooler_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  cooler_type VARCHAR(10) NOT NULL,
  height_mm DECIMAL(8,2),
  radiator_size_mm INT,
  radiator_thickness_mm DECIMAL(8,2),
  cooling_assessment JSON,
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (cooler_type IN ('AIR','AIO','CUSTOM')),
  CHECK (height_mm > 0 AND radiator_size_mm > 0)
) ENGINE=InnoDB;
CREATE TABLE fan_specs (
  variant_id BIGINT NOT NULL PRIMARY KEY,
  size_mm INT NOT NULL,
  thickness_mm DECIMAL(8,2),
  connector_type VARCHAR(30),
  voltage_v DECIMAL(6,2),
  current_a DECIMAL(6,3),
  source_id BIGINT,
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id),
  CHECK (size_mm > 0)
) ENGINE=InnoDB;
CREATE TABLE motherboard_storage_slots (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  motherboard_variant_id BIGINT NOT NULL,
  slot_code VARCHAR(40) NOT NULL,
  interface_type VARCHAR(30) NOT NULL,
  supported_protocols JSON NOT NULL,
  supported_form_factors JSON NOT NULL,
  pcie_generation SMALLINT,
  pcie_lanes SMALLINT,
  UNIQUE KEY uk_mb_storage_slot (motherboard_variant_id, slot_code),
  FOREIGN KEY (motherboard_variant_id) REFERENCES motherboard_specs(variant_id)
) ENGINE=InnoDB;
CREATE TABLE motherboard_expansion_slots (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  motherboard_variant_id BIGINT NOT NULL,
  slot_code VARCHAR(40) NOT NULL,
  physical_lanes SMALLINT NOT NULL,
  electrical_lanes SMALLINT NOT NULL,
  pcie_generation SMALLINT,
  UNIQUE KEY uk_mb_expansion_slot (motherboard_variant_id, slot_code),
  FOREIGN KEY (motherboard_variant_id) REFERENCES motherboard_specs(variant_id),
  CHECK (physical_lanes > 0 AND electrical_lanes > 0 AND electrical_lanes <= physical_lanes)
) ENGINE=InnoDB;
CREATE TABLE motherboard_lane_sharing_rules (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  motherboard_variant_id BIGINT NOT NULL,
  condition_config JSON NOT NULL,
  effect_config JSON NOT NULL,
  description VARCHAR(1000) NOT NULL,
  source_id BIGINT,
  FOREIGN KEY (motherboard_variant_id) REFERENCES motherboard_specs(variant_id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id)
) ENGINE=InnoDB;
CREATE TABLE case_supported_form_factors (
  case_variant_id BIGINT NOT NULL,
  component_type VARCHAR(20) NOT NULL,
  form_factor VARCHAR(30) NOT NULL,
  PRIMARY KEY (case_variant_id, component_type, form_factor),
  FOREIGN KEY (case_variant_id) REFERENCES case_specs(variant_id),
  CHECK (component_type IN ('MOTHERBOARD','PSU'))
) ENGINE=InnoDB;
CREATE TABLE case_mount_points (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  case_variant_id BIGINT NOT NULL,
  position_code VARCHAR(40) NOT NULL,
  device_type VARCHAR(20) NOT NULL,
  size_code VARCHAR(30) NOT NULL,
  quantity SMALLINT NOT NULL,
  UNIQUE KEY uk_case_mount (case_variant_id, position_code, device_type, size_code),
  FOREIGN KEY (case_variant_id) REFERENCES case_specs(variant_id),
  CHECK (device_type IN ('FAN','RADIATOR','STORAGE')),
  CHECK (quantity > 0)
) ENGINE=InnoDB;
CREATE TABLE case_clearance_profiles (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  case_variant_id BIGINT NOT NULL,
  name VARCHAR(150) NOT NULL,
  installation_conditions JSON NOT NULL,
  clearance_limits JSON NOT NULL,
  source_id BIGINT,
  FOREIGN KEY (case_variant_id) REFERENCES case_specs(variant_id),
  FOREIGN KEY (source_id) REFERENCES spec_sources(id)
) ENGINE=InnoDB;
CREATE TABLE cooler_supported_sockets (
  cooler_variant_id BIGINT NOT NULL,
  socket_id BIGINT NOT NULL,
  requires_mounting_kit BOOLEAN NOT NULL DEFAULT FALSE,
  mounting_kit_variant_id BIGINT,
  PRIMARY KEY (cooler_variant_id, socket_id),
  FOREIGN KEY (cooler_variant_id) REFERENCES cooler_specs(variant_id),
  FOREIGN KEY (socket_id) REFERENCES cpu_sockets(id),
  FOREIGN KEY (mounting_kit_variant_id) REFERENCES product_variants(id)
) ENGINE=InnoDB;
CREATE TABLE power_connector_types (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(40) NOT NULL UNIQUE,
  name VARCHAR(100) NOT NULL
) ENGINE=InnoDB;
CREATE TABLE variant_power_connectors (
  variant_id BIGINT NOT NULL,
  connector_type_id BIGINT NOT NULL,
  direction VARCHAR(10) NOT NULL,
  quantity SMALLINT NOT NULL,
  PRIMARY KEY (variant_id, connector_type_id, direction),
  FOREIGN KEY (variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (connector_type_id) REFERENCES power_connector_types(id),
  CHECK (direction IN ('SUPPLY','REQUIRE')),
  CHECK (quantity > 0)
) ENGINE=InnoDB;
CREATE TABLE variant_included_items (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  parent_variant_id BIGINT NOT NULL,
  included_variant_id BIGINT,
  item_type VARCHAR(30) NOT NULL,
  name VARCHAR(255) NOT NULL,
  quantity SMALLINT NOT NULL DEFAULT 1,
  specifications JSON,
  FOREIGN KEY (parent_variant_id) REFERENCES product_variants(id),
  FOREIGN KEY (included_variant_id) REFERENCES product_variants(id),
  CHECK (quantity > 0),
  CHECK (included_variant_id IS NULL OR included_variant_id <> parent_variant_id)
) ENGINE=InnoDB;
CREATE TABLE compatibility_rules (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,

  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  promotion_id BIGINT NOT NULL,
  version_no INT NOT NULL,
  starts_at DATETIME(6) NOT NULL,
  ends_at DATETIME(6) NOT NULL,
  priority INT NOT NULL DEFAULT 0,
  stacking_policy VARCHAR(20) NOT NULL DEFAULT 'EXCLUSIVE',
  stacking_group VARCHAR(60),
  threshold_basis VARCHAR(30) NOT NULL DEFAULT 'ELIGIBLE_GROSS_PRE_DISCOUNT',
  currency_code CHAR(3) NOT NULL DEFAULT 'VND',
  budget_limit DECIMAL(19,2),
  global_usage_limit INT,
  per_customer_limit INT,
  max_applications_per_order INT NOT NULL DEFAULT 1,
  requires_coupon BOOLEAN NOT NULL DEFAULT FALSE,
  rules_schema_version INT NOT NULL DEFAULT 1,
  published_at DATETIME(6),
  UNIQUE KEY uk_promotion_version (promotion_id, version_no),
  KEY ix_promotion_window (starts_at, ends_at),
  FOREIGN KEY (promotion_id) REFERENCES promotions(id),
  CHECK (version_no > 0 AND ends_at > starts_at),
  CHECK (stacking_policy IN ('EXCLUSIVE','STACKABLE','BEST_OF_GROUP')),
  CHECK (stacking_policy <> 'BEST_OF_GROUP' OR stacking_group IS NOT NULL),
  CHECK (threshold_basis IN ('ELIGIBLE_GROSS_PRE_DISCOUNT','ELIGIBLE_GROSS_POST_ITEM')),
  CHECK (budget_limit >= 0 AND global_usage_limit > 0 AND per_customer_limit > 0),
  CHECK (max_applications_per_order > 0)
) ENGINE=InnoDB;
CREATE TABLE promotion_condition_groups (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  promotion_version_id BIGINT NOT NULL,
  parent_group_id BIGINT,
  logical_operator VARCHAR(3) NOT NULL DEFAULT 'AND',
  root_version_id BIGINT GENERATED ALWAYS AS (CASE WHEN parent_group_id IS NULL THEN promotion_version_id ELSE NULL END) STORED,
  UNIQUE KEY uk_promotion_root (root_version_id),
  UNIQUE KEY uk_condition_group_pair (promotion_version_id, id),
  FOREIGN KEY (promotion_version_id) REFERENCES promotion_versions(id),
  FOREIGN KEY (promotion_version_id, parent_group_id) REFERENCES promotion_condition_groups(promotion_version_id, id),
  CHECK (logical_operator IN ('AND','OR'))
) ENGINE=InnoDB;
CREATE TABLE promotion_conditions (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,

  condition_type VARCHAR(30) NOT NULL,
  comparison_operator VARCHAR(15) NOT NULL,
  configuration JSON NOT NULL,

  CHECK (comparison_operator IN ('EQ','NE','GT','GTE','LT','LTE','IN','NOT_IN','BETWEEN'))
) ENGINE=InnoDB;
CREATE TABLE promotion_actions (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,

  sort_order INT NOT NULL DEFAULT 0,
  FOREIGN KEY (promotion_version_id) REFERENCES promotion_versions(id),
  CHECK (action_type IN ('FIXED_DISCOUNT','PERCENT_DISCOUNT','GIFT','FREE_SHIPPING','SHIPPING_DISCOUNT')),
  CHECK (amount >= 0 AND percent_value BETWEEN 0 AND 100 AND max_discount >= 0 AND gift_quantity > 0),
  CHECK (action_type NOT IN ('FIXED_DISCOUNT','SHIPPING_DISCOUNT') OR amount IS NOT NULL),
  CHECK (action_type <> 'PERCENT_DISCOUNT' OR percent_value IS NOT NULL),
  CHECK (action_type <> 'GIFT' OR gift_quantity IS NOT NULL)
) ENGINE=InnoDB;
CREATE TABLE promotion_targets (

-- ================================================================
-- 07. CART / QUOTE / ORDER
-- Prices are tax-inclusive. tax_amount is informational, not added twice.
-- total = gross items - item/order discounts + net shipping + net services.
-- ================================================================
CREATE TABLE carts (
  id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT,
  guest_token_hash BINARY(32),
  status VARCHAR(15) NOT NULL DEFAULT 'ACTIVE',
  currency_code CHAR(3) NOT NULL DEFAULT 'VND',
  expires_at DATETIME(6),
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  version BIGINT NOT NULL DEFAULT 0,
  FOREIGN KEY (user_id) REFERENCES users(id),
  KEY ix_cart_owner (user_id, status),
  KEY ix_cart_guest (guest_token_hash, status),
  CHECK ((user_id IS NOT NULL) + (guest_token_hash IS NOT NULL) = 1),
  CHECK (status IN ('ACTIVE','CONVERTED','ABANDONED'))

-- ================================================================
-- 12. REPORTING VIEWS
-- Report date uses UTC+07:00. Monetary dimensions always include currency.
-- ================================================================
CREATE VIEW v_sellable_inventory AS
SELECT b.variant_id, SUM(b.on_hand) AS on_hand,
       SUM(b.reserved) AS reserved, SUM(b.available) AS available
FROM inventory_balances b
JOIN warehouses w ON w.id = b.warehouse_id AND w.is_active = 1
WHERE b.stock_condition = 'SELLABLE'
GROUP BY b.variant_id;

CREATE VIEW v_product_ratings AS
SELECT product_id, COUNT(*) AS review_count, ROUND(AVG(rating), 2) AS average_rating
FROM reviews WHERE status = 'APPROVED' GROUP BY product_id;

CREATE VIEW v_daily_completed_orders AS
SELECT DATE(CONVERT_TZ(completed_at, '+00:00', '+07:00')) AS report_date,
       currency_code, COUNT(*) AS completed_orders,
       SUM(subtotal) AS gross_item_amount, SUM(discount_amount) AS item_discount,
       SUM(grand_total) AS completed_order_value
FROM orders
WHERE order_status = 'COMPLETED' AND completed_at IS NOT NULL
GROUP BY DATE(CONVERT_TZ(completed_at, '+00:00', '+07:00')), currency_code;

-- Cash movement, NOT accrual accounting revenue. Refund recorded on refund date.
CREATE VIEW v_daily_cash_flow AS
SELECT report_date, currency_code, SUM(cash_in) AS cash_in,
       SUM(cash_out) AS cash_out, SUM(cash_in) - SUM(cash_out) AS net_cash
FROM (
  SELECT DATE(CONVERT_TZ(t.occurred_at, '+00:00', '+07:00')) AS report_date,
         p.currency_code, t.amount AS cash_in, CAST(0 AS DECIMAL(19,2)) AS cash_out
  FROM payment_transactions t JOIN payments p ON p.id = t.payment_id
  WHERE t.status = 'SUCCEEDED' AND t.transaction_type IN ('CAPTURE','SALE','COD_COLLECT')
  UNION ALL
  SELECT DATE(CONVERT_TZ(r.completed_at, '+00:00', '+07:00')),
         p.currency_code, CAST(0 AS DECIMAL(19,2)), r.amount
  FROM refunds r JOIN payments p ON p.id = r.payment_id
  WHERE r.status = 'SUCCEEDED' AND r.completed_at IS NOT NULL
) cash_movements
GROUP BY report_date, currency_code;

CREATE VIEW v_best_selling_variants AS
SELECT oi.variant_id, o.currency_code, SUM(oi.quantity) AS sold_quantity,
       SUM(oi.line_total) AS item_sales_before_returns,
       SUM(oi.quantity * oi.unit_cost) AS recorded_cost_before_returns
FROM order_items oi JOIN orders o ON o.id = oi.order_id
WHERE o.order_status = 'COMPLETED' AND oi.is_gift = 0
GROUP BY oi.variant_id, o.currency_code;

-- Cohort conversion: session start day; numerator = sessions with a confirmed order.
CREATE VIEW v_daily_session_conversion AS
SELECT DATE(CONVERT_TZ(s.started_at, '+00:00', '+07:00')) AS cohort_date,
       COUNT(*) AS sessions,
       SUM(CASE WHEN converted.session_id IS NOT NULL THEN 1 ELSE 0 END) AS converted_sessions,
       ROUND(100.0 * SUM(CASE WHEN converted.session_id IS NOT NULL THEN 1 ELSE 0 END) / COUNT(*), 2) AS conversion_percent
FROM analytics_sessions s
LEFT JOIN (
  SELECT DISTINCT e.session_id
  FROM analytics_events e JOIN orders o ON o.id = e.order_id
  WHERE e.event_type = 'PURCHASE' AND o.order_status IN ('CONFIRMED','PROCESSING','COMPLETED')
) converted ON converted.session_id = s.id
WHERE s.is_bot = 0 AND s.consent_status IN ('GRANTED','NOT_REQUIRED')
GROUP BY DATE(CONVERT_TZ(s.started_at, '+00:00', '+07:00'));
